5. CODE NODE "Định dạng nội dung FB" (bản đang chạy)
--------------------------------------------------------------------------------
const raw = $input.first().json.data || '';
const wraps = String(raw).split('class="tgme_widget_message_wrap');
if (wraps.length < 2) return [];
const block = wraps[wraps.length - 1];

// --- 1. message_id cua bai dang trong kenh Telegram ---
let message_id = 0;
const mId = block.match(/t\.me\/[^"'\s]+\/(\d+)/);
if (mId) message_id = parseInt(mId[1], 10);

// --- 2. Noi dung text ---
const parts = block.split('class="tgme_widget_message_text');
if (parts.length < 2) return [];
let seg = parts[1];
seg = seg.slice(seg.indexOf('>') + 1);
seg = seg.slice(0, seg.indexOf('</div>'));

let text = seg
  .replace(/<br\s*\/?>/gi, '\n')
  .replace(/<[^>]+>/g, '')
  .replace(/&quot;/g, '"')
  .replace(/&#39;/g, "'")
  .replace(/&amp;/g, '&')
  .replace(/&lt;/g, '<')
  .replace(/&gt;/g, '>')
  .replace(/&nbsp;/g, ' ')
  .replace(/\n{3,}/g, '\n\n')
  .trim();

const cut = text.indexOf('======');
if (cut > 0) text = text.slice(0, cut).trim();
if (!text) return [];

// --- 3. Anh cua bai (photo hoac thumbnail video) ---
let photo_url = '';
const mPhoto = block.match(/background-image:url\('(https:\/\/cdn\d*\.telesco\.pe\/[^']+)'\)/);
if (mPhoto) photo_url = mPhoto[1];

// --- 4. Chong dang trung: 100 message_id gan nhat + hash noi dung (FNV-1a) ---
const store = $getWorkflowStaticData('global');
const postedIds = Array.isArray(store.postedIds) ? store.postedIds : [];

function stableHash(str) {
  let h = 2166136261 >>> 0;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h.toString(16);
}

const contentHash = stableHash(text + '|' + photo_url);

if (message_id > 0 && postedIds.includes(message_id)) return [];
if (store.lastContentHash === contentHash) return [];

if (message_id > 0) {
  postedIds.push(message_id);
  while (postedIds.length > 100) postedIds.shift();
  store.postedIds = postedIds;
}
store.lastContentHash = contentHash;
store.lastMessageId = message_id;
store.lastQuote = text;

const cta = '\n\n' +
  'Đừng học thêm một mô hình nến nữa. Bạn đã học đủ rồi. Vấn đề nằm ở chỗ khác, và nó ở đây → nghetrading.com/bai-viet.html\n' +
  'Đã đến lúc coi trading là 1 nghề nghiệp thật sự nghiêm túc.';

return [{ fb_content: text + cta, photo_url, message_id, content_hash: contentHash }];
